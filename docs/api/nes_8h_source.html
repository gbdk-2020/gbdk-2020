1046inline void move_bkg(scroll_x_t x, scroll_y_t y) {
1047 // store low 8 bits to shadow scroll registers
1048 bkg_scroll_x = (uint8_t)x;
1049 bkg_scroll_y = (uint8_t)(y >= 240 ? (y - 240) : y);
1050 // store 9th bit of x and y in shadow PPUCTRL register
1051#if DEVICE_SCREEN_BUFFER_WIDTH > 32 && DEVICE_SCREEN_BUFFER_HEIGHT > 30
1052 uint8_t msb_x = (uint8_t)((x >> 8) & 1);
1053 uint8_t msb_y = (uint8_t)(y >= 240 ? 1 : 0);
1054 shadow_PPUCTRL = (shadow_PPUCTRL & 0xFC) | (msb_y << 1) | msb_x;
1055#elif DEVICE_SCREEN_BUFFER_WIDTH > 32
1056 uint8_t msb_x = (uint8_t)((x >> 8) & 1);
1057 shadow_PPUCTRL = (shadow_PPUCTRL & 0xFC) | msb_x;
1058#elif DEVICE_SCREEN_BUFFER_HEIGHT > 30
1059 uint8_t msb_y = (uint8_t)(y >= 240 ? 1 : 0);
1060 shadow_PPUCTRL = (shadow_PPUCTRL & 0xFC) | (msb_y << 1);
1061#endif
1062}
